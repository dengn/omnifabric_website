from omnifabric import Client

of = Client(api_key="of_live_...")

# hybrid search + an agent call, one SDK
docs = of.search(
    query="refund policy",
    top_k=8, hybrid=True,
)
answer = of.agent("support").run(
    input=question, context=docs,
)  # lineage captured automatically
